Step 2 — FastAPI 서버 신규 작성

기존 supertonic serve가 노출하던 엔드포인트를 동일하게 구현 → player.py 변경 없이 드롭인 교체.

# tts_server/supertonic_mlx_server.py
from __future__ import annotations
import io, asyncio
from pathlib import Path
import soundfile as sf
from fastapi import FastAPI, HTTPException
from fastapi.responses import Response
from pydantic import BaseModel
from supertonic_mlx import SupertonicMLX

MODEL_DIR = Path.home() / ".cache" / "supertonic3-mlx"

app = FastAPI()
_model: SupertonicMLX | None = None
_lock = asyncio.Lock()

@app.on_event("startup")
async def _load():
    global _model
    _model = SupertonicMLX(MODEL_DIR)
    # Metal GPU 워밍업 (첫 추론 ~2s 컴파일)
    style = _model.get_voice_style("M1")
    _model.synthesize("워밍업.", "ko", style, total_step=2)

class TTSRequest(BaseModel):
    text: str
    lang: str = "ko"
    voice: str = "M1"
    steps: int = 8
    speed: float = 1.05

@app.get("/v1/health")
async def health():
    return {"status": "ok"}

@app.post("/v1/tts")
async def tts(req: TTSRequest):
    if _model is None:
        raise HTTPException(503, "모델 로딩 중")
    async with _lock:            # Metal GPU 단일 스레드 직렬화
        style = _model.get_voice_style(req.voice)
        wav, _ = _model.synthesize(req.text, req.lang, style,
                                   total_step=req.steps, speed=req.speed)
    buf = io.BytesIO()
    sf.write(buf, wav[0], _model.sample_rate, format="WAV")
    return Response(buf.getvalue(), media_type="audio/wav")

Step 3 — supervisor.py 수정 (1함수)

# tts_server/supervisor.py 수정 — _start_supertonic() 1함수만 교체
def _start_supertonic() -> subprocess.Popen:
    with open("/tmp/supertonic.log", "a") as log:
        return subprocess.Popen(
            [str(VENV_BIN / "uvicorn"),
             "tts_server.supertonic_mlx_server:app",
             "--host", "127.0.0.1",
             "--port", "7788",
             "--workers", "1"],  # Metal GPU는 단일 프로세스 필수
            stdout=log, stderr=log,
        )

주의사항

항목내용
Metal GPU 단일 스레드MLX Metal 스트림은 스레드 이동 불가. asyncio.Lock + --workers 1 필수.
첫 추론 워밍업첫 synthesize 호출 시 Metal 컴파일 ~2s. startup 시 더미 텍스트로 선행 워밍업 권장.
HF_HUB_OFFLINE환경변수 HF_HUB_OFFLINE=1 설정 시 모델 다운로드 불가. 다운로드 시 임시 해제 필요.
HMG SSL 프록시HuggingFace 다운로드 시 SSL 인터셉트 오류 가능. REQUESTS_CA_BUNDLE 또는 verify=False 처리 필요.
ailuntx/supertonic-mlx 성숙도Star 0, 2026-05-30 첫 공개. ONNX 대비 오차 ~1e-5 검증됨. 프로덕션 안정성은 직접 검증 권장.
