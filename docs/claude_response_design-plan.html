4.1 CanonicalEvent

from dataclasses import dataclass, field
from enum import Enum
import uuid

class Severity(str, Enum):
    CRITICAL = "critical"
    HIGH     = "high"
    MEDIUM   = "medium"
    LOW      = "low"
    INFO     = "info"

class InterruptPolicy(str, Enum):
    ALWAYS   = "always"     # 즉시 끼어들기
    QUEUE    = "queue"      # 현재 발화 후 재생
    DISCARD  = "discard"    # 조용히 버림

@dataclass(frozen=True)
class LanguageProfile:
    primary: str         # "ko" | "en" | "mixed"
    confidence: float    # 0.0–1.0
    script: str          # "hangul" | "latin" | "mixed"
    mixed_content: bool

@dataclass
class CanonicalEvent:
    event_id:        str   = field(default_factory=lambda: str(uuid.uuid4()))
    idempotency_key: str   = ""
    fingerprint:     str   = ""
    dedupe_key:      str   = ""
    correlation_id:  str   = ""
    source:          str   = ""          # "coding_agent" | "grafana"
    source_event_type: str = ""          # "stop" | "subagent_stop" | "alert_firing"
    severity:        Severity = Severity.INFO
    priority_score:  int   = 0           # 0–100 (높을수록 먼저)
    raw_text:        str   = ""
    language:        LanguageProfile | None = None
    ttl:             float = 30.0        # 초
    interrupt_policy: InterruptPolicy = InterruptPolicy.QUEUE
    version:         int   = 1
    metadata:        dict  = field(default_factory=dict)
    trace_id:        str   = ""

4.3 SpeechPipeline

# speech/pipeline.py
from dataclasses import dataclass
from typing import Callable

@dataclass
class SpeechContext:
    text: str
    language: str = "ko"
    segments: list[str] = field(default_factory=list)
    ssml: str = ""
    metadata: dict = field(default_factory=dict)

Stage = Callable[[SpeechContext], SpeechContext]

STAGE_ORDER = [
    "lang_detect",          # 1. 언어 감지 (global<5ms + sliding window)
    "segment_splitter",     # 2. 문장 분할
    "identifier_normalizer",# 3. snake_case / CamelCase / UUID 처리
    "pattern_suppressor",   # 4. URL·코드블록·마크다운 억제
    "pronunciation_dict",   # 5. IT 용어 발음 변환 (Trie + LRU 10K)
    "normalization",        # 6. 숫자·단위·특수문자 정규화
    "prosody_boundary",     # 7. 언어 경계 Break Time 주입
    "ssml_abstraction",     # 8. SSML 태그 생성
]

class SpeechPipeline:
    def __init__(self, stages: list[Stage]) -> None:
        self._stages = stages

    async def process(self, text: str) -> SpeechContext:
        ctx = SpeechContext(text=text)
        for stage in self._stages:
            ctx = await stage(ctx) if asyncio.iscoroutinefunction(stage) else stage(ctx)
        return ctx
